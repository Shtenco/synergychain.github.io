cmake_minimum_required(VERSION 3.20)
project(MidasExactZlib C)
if(NOT DEFINED ZLIB_SOURCE_DIR OR NOT DEFINED ZLIB_BUILD_DIR)
  message(FATAL_ERROR "ZLIB_SOURCE_DIR and ZLIB_BUILD_DIR are required")
endif()
add_library(MidasExactZlib SHARED MidasExactZlib.c)
target_include_directories(MidasExactZlib PRIVATE "${ZLIB_SOURCE_DIR}" "${ZLIB_BUILD_DIR}")
target_link_libraries(MidasExactZlib PRIVATE "${ZLIB_BUILD_DIR}/Release/zlibstatic.lib")
target_compile_definitions(MidasExactZlib PRIVATE _CRT_SECURE_NO_WARNINGS)
set_target_properties(MidasExactZlib PROPERTIES OUTPUT_NAME "MidasExactZlib")
